# ============================================================
#  EXPERIMENT 2 : Recovery of Deleted Files
#  Recovery with PhotoRec (signature based carving)
# ============================================================
export PATH="$PWD/tools/bin:$PATH"
cd artifacts/exp2

# --- clean the previous carving run ---
rm -rf carved.* photorec.log

# --- carve the deleted files out of the raw image ---
# PhotoRec is a curses program. It is started in its own session (setsid) with
# the terminal detached, so its screen escapes cannot overwrite this terminal;
# the session report is read back from photorec.log.
setsid photorec /log /d carved /cmd usb_drive.img partition_none,fileopt,everything,enable,search < /dev/null > /dev/null 2>&1

echo "----- PhotoRec session summary (photorec.log) -----"
grep -E 'recovered|Total|exited normally' photorec.log | tail -6

echo "----- the evidence files that were carved out -----"
ls -l carved.1/*.txt carved.1/*.jpg

# --- match every carved file against the hashes taken before deletion ---
echo "----- hash comparison -----"
### BLOCK
for f in $(find carved.1 -type f \( -name '*.txt' -o -name '*.jpg' \) | sort); do
    h=$(sha256sum "$f" | cut -d' ' -f1)
    if grep -q "$h" hashes_before.txt; then
        printf 'MATCH     %-18s == %s\n' "$(basename "$f")" "$(awk -v h="$h" '$1==h{print $2}' hashes_before.txt)"
    else
        printf 'NO MATCH  %s\n' "$(basename "$f")"
    fi
done
### ENDBLOCK
